          "ก. ฮิสโทแกรม (Histogram)",
          "ข. รูปหลายเหลี่ยมของความถี่ (Frequency Polygon)",
          "ค. เส้นโค้งของความถี่ (Frequency Curve)",
          "ง. ตารางแจกแจงความถี่สะสม (Cumulative Frequency Table)"
        ],
        correctIndex: 0,
        explanation: [
          "<strong>หลักการคิด:</strong> การแสดงการแจกแจงความถี่ของข้อมูลด้วยแผนภูมิแท่งเรียงชิดติดกัน โดยแกนนอนแทนขอบเขตของอันตรภาคชั้น และแกนตั้งแทนความถี่ของข้อมูล เรียกว่า <strong>ฮิสโทแกรม (Histogram)</strong> (อ้างอิง ตอนที่ 2 หน้า 21)"
        ]
      }
    ];

    // State
    let userAnswers = {};
    let isSubmitted = false;
    let currentFilter = "all"; // 'all' | 'wrong'
    
    // Timer configurations
    const QUIZ_TIME_LIMIT_SECONDS = 30 * 60; // 15 minutes
    let timeLeft = QUIZ_TIME_LIMIT_SECONDS;
    let timerInterval = null;
    let timeSpentSeconds = 0;

    // Elements
    const questionsContainer = document.getElementById("questionsContainer");
    const progressText = document.getElementById("progressText");
    const progressBar = document.getElementById("progressBar");
    const progressFill = document.getElementById("progressFill");
    const quizForm = document.getElementById("quizForm");
    const validationBanner = document.getElementById("validationBanner");
    const scoreBanner = document.getElementById("scoreBanner");
    const submitBtn = document.getElementById("submitBtn");
    const clearBtn = document.getElementById("clearBtn");
    const retakeBtn = document.getElementById("retakeBtn");
    const filterAllBtn = document.getElementById("filterAllBtn");
    const filterWrongBtn = document.getElementById("filterWrongBtn");
    const timerWrap = document.getElementById("timerWrap");
    const timerLabel = document.getElementById("timerLabel");

    const studentNameInput = document.getElementById("studentName");
    const studentClassInput = document.getElementById("studentClass");
    const studentNoInput = document.getElementById("studentNo");

    function renderQuestions() {
      questionsContainer.innerHTML = "";

      QUESTIONS.forEach((q, idx) => {
        const card = document.createElement("section");
        card.className = "card question-card";
        card.id = `question-card-${q.id}`;
        card.dataset.qid = q.id;

        let extraContentHTML = "";
        if (q.dataHighlight) {
          extraContentHTML += `<div class="data-highlight-box">${q.dataHighlight}</div>`;
        }
        if (q.tableData) {
          const rowsHTML = q.tableData.map(row => `
            <tr>
              <td>${row.interval}</td>
              <td>${row.freq}</td>
              <td>${row.highlight ? `<span class="highlight-var">${row.cum}</span>` : row.cum}</td>
            </tr>
          `).join("");
          extraContentHTML += `
            <div class="table-wrapper">
              <table class="freq-table">
                <thead>
                  <tr>
                    <th>อันตรภาคชั้น</th>
                    <th>ความถี่ (f)</th>
                    <th>ความถี่สะสม</th>
                  </tr>
                </thead>
                <tbody>${rowsHTML}</tbody>
              </table>
            </div>
          `;
        }

        const choicesHTML = q.choices.map((choiceText, cIdx) => `
          <label class="choice-label" id="label-q${q.id}-c${cIdx}">
            <input
              type="radio"
              class="choice-radio"
              name="question_${q.id}"
              value="${cIdx}"
              data-qid="${q.id}"
              data-cidx="${cIdx}"
            >
            <span class="choice-text">${choiceText}</span>
            <span class="choice-feedback-tag" id="tag-q${q.id}-c${cIdx}" style="display:none;"></span>
          </label>
        `).join("");

        const explanationListHTML = q.explanation.map(item => `<li>${item}</li>`).join("");

        card.innerHTML = `
          <div class="question-header">
            <h3 class="question-title">ข้อที่ ${idx + 1}. ${q.question} <span class="req-star">*</span></h3>
            <div class="question-meta">
              <span class="source-tag">${q.source}</span>
              <span class="point-tag">1 คะแนน</span>
              <span class="status-badge" id="status-badge-${q.id}"></span>
            </div>
          </div>
          ${extraContentHTML}
          <div class="choices-list" role="radiogroup" aria-label="ตัวเลือกสำหรับข้อที่ ${idx + 1}">
            ${choicesHTML}
          </div>
          <div class="explanation-box" id="explanation-${q.id}">
            <div class="explanation-header">เฉลยและวิธีคิดอย่างละเอียด:</div>
            <ul class="explanation-steps">
              ${explanationListHTML}
            </ul>
          </div>
        `;

        questionsContainer.appendChild(card);
      });

      // Attach listeners to radio inputs
      const radios = questionsContainer.querySelectorAll(".choice-radio");
      radios.forEach(radio => {
        radio.addEventListener("change", (e) => {
          if (isSubmitted) return;
          const qid = parseInt(e.target.dataset.qid, 10);
          const cidx = parseInt(e.target.dataset.cidx, 10);
          userAnswers[qid] = cidx;

          // Update visual selection state for this question
          const card = document.getElementById(`question-card-${qid}`);
          card.classList.remove("unanswered-error");
          const labels = card.querySelectorAll(".choice-label");
          labels.forEach((lbl, i) => {
            lbl.classList.toggle("selected", i === cidx);
          });

          updateProgress();
        });
      });

      // Add click support directly to the label wrapper for absolute touch/click reliability
      const labels = questionsContainer.querySelectorAll(".choice-label");
      labels.forEach(lbl => {
        lbl.addEventListener("click", (e) => {
          if (isSubmitted) return;
          const radio = lbl.querySelector(".choice-radio");
          if (!radio || radio.disabled) return;
          
          // If user clicked outside the actual radio input element, trigger a click on the radio
          if (e.target !== radio) {
            e.preventDefault();
            radio.click();
          }
        });
      });
    }

    function updateProgress() {
      const answeredCount = Object.keys(userAnswers).length;
      const total = QUESTIONS.length;
      const pct = Math.round((answeredCount / total) * 100);
      progressText.textContent = `ตอบแล้ว ${answeredCount} / ${total} ข้อ`;
      progressBar.setAttribute("aria-valuenow", String(answeredCount));
      progressFill.style.width = `${pct}%`;
    }

    function validateField(inputEl, errorEl) {
      const val = inputEl.value.trim();
      const valid = val.length > 0;
      inputEl.setAttribute("aria-invalid", valid ? "false" : "true");
      errorEl.classList.toggle("visible", !valid);
      return valid;
    }

    // Timer functions
    function startTimer() {
      clearInterval(timerInterval);
      timeLeft = QUIZ_TIME_LIMIT_SECONDS;
      timeSpentSeconds = 0;
      updateTimerDisplay();

      timerInterval = setInterval(() => {
        if (isSubmitted) {
          clearInterval(timerInterval);
          return;
        }

        timeLeft--;
        timeSpentSeconds++;
        updateTimerDisplay();

        if (timeLeft <= 0) {
          clearInterval(timerInterval);
          handleTimeout();
        }
      }, 1000);
    }

    function updateTimerDisplay() {
      const minutes = Math.floor(timeLeft / 60);
      const seconds = timeLeft % 60;
      const displayStr = `${String(minutes).padStart(2, '0')}:${String(seconds).padStart(2, '0')}`;
      
      timerLabel.textContent = `เวลาที่เหลือ: ${displayStr} น.`;

      if (timeLeft <= 120) { // Warning: Less than or equal to 2 mins left
        timerWrap.classList.add("warning");
      } else {
        timerWrap.classList.remove("warning");
      }
    }

    function handleTimeout() {
      // Automatically lock and grade the quiz
      validationBanner.innerHTML = "<strong>หมดเวลาทำข้อสอบแล้ว!</strong> ระบบได้ทำการบันทึกและส่งคำตอบปัจจุบันของคุณอัตโนมัติ";
      validationBanner.classList.add("visible");
      
      // Auto-set mock values for empty student info if blank
      if (!studentNameInput.value.trim()) studentNameInput.value = "ผู้เข้าสอบหมดเวลา";
      if (!studentClassInput.value.trim()) studentClassInput.value = "ไม่ได้ระบุ";
      if (!studentNoInput.value.trim()) studentNoInput.value = "-";
      
      gradeQuiz();
    }

    [
      [studentNameInput, document.getElementById("nameError")],
      [studentClassInput, document.getElementById("classError")],
      [studentNoInput, document.getElementById("noError")]
    ].forEach(([inputEl, errorEl]) => {
      inputEl.addEventListener("blur", () => {
        if (!isSubmitted) validateField(inputEl, errorEl);
      });
      inputEl.addEventListener("input", () => {
        if (inputEl.getAttribute("aria-invalid") === "true") {
          validateField(inputEl, errorEl);
        }
      });
    });

    quizForm.addEventListener("submit", (e) => {
      e.preventDefault();
      if (isSubmitted) return;

      const nameOk = validateField(studentNameInput, document.getElementById("nameError"));
      const classOk = validateField(studentClassInput, document.getElementById("classError"));
      const noOk = validateField(studentNoInput, document.getElementById("noError"));

      const unansweredIds = [];
      QUESTIONS.forEach(q => {
        const card = document.getElementById(`question-card-${q.id}`);
        if (userAnswers[q.id] === undefined) {
          unansweredIds.push(q.id);
          card.classList.add("unanswered-error");
        } else {
          card.classList.remove("unanswered-error");
        }
      });

      if (!nameOk || !classOk || !noOk || unansweredIds.length > 0) {
        const messages = [];
        if (!nameOk || !classOk || !noOk) {
          messages.push("กรุณากรอกข้อมูลผู้เข้าสอบ (ชื่อ-นามสกุล, ระดับชั้น, รหัสนักศึกษา) ให้ครบถ้วน");
        }
        if (unansweredIds.length > 0) {
          messages.push(`ยังไม่ได้เลือกคำตอบในข้อที่: ${unansweredIds.join(", ")} (กรุณาทำข้อสอบให้ครบทั้ง 10 ข้อก่อนส่ง)`);
        }
        validationBanner.innerHTML = messages.join("<br>");
        validationBanner.classList.add("visible");

        // Scroll to the first error
        if (!nameOk || !classOk || !noOk) {
          document.getElementById("studentInfoCard").scrollIntoView({ behavior: "smooth", block: "center" });
        } else if (unansweredIds.length > 0) {
          document.getElementById(`question-card-${unansweredIds[0]}`).scrollIntoView({ behavior: "smooth", block: "center" });
        }
        return;
      }

      validationBanner.classList.remove("visible");
      gradeQuiz();
    });

    function gradeQuiz() {
      isSubmitted = true;
      clearInterval(timerInterval);
      let score = 0;

      // Lock student inputs
      studentNameInput.disabled = true;
      studentClassInput.disabled = true;
      studentNoInput.disabled = true;

      QUESTIONS.forEach(q => {
        const selectedIdx = userAnswers[q.id];
        const isCorrect = selectedIdx === q.correctIndex;
        if (isCorrect) score++;

        const card = document.getElementById(`question-card-${q.id}`);
        card.classList.add(isCorrect ? "graded-correct" : "graded-incorrect");
        card.dataset.correct = isCorrect ? "true" : "false";

        // Status badge
        const badge = document.getElementById(`status-badge-${q.id}`);
        badge.className = `status-badge ${isCorrect ? "correct" : "incorrect"}`;
        badge.textContent = isCorrect ? "✓ ตอบถูก (+1 คะแนน)" : "✕ ตอบผิด (0 คะแนน)";

        // Highlight choices & lock radios
        q.choices.forEach((_, cIdx) => {
          const label = document.getElementById("label-q" + q.id + "-c" + cIdx);
          const tag = document.getElementById("tag-q" + q.id + "-c" + cIdx);
          const radio = label.querySelector(".choice-radio");
          if (radio) radio.disabled = true;
          label.classList.add("disabled");
          label.classList.remove("selected");

          if (cIdx === selectedIdx) {
            if (isCorrect) {
              label.classList.add("correct-choice");
              tag.style.display = "inline-block";
              tag.className = "choice-feedback-tag tag-correct";
              tag.textContent = "คำตอบของคุณ (ถูกต้อง)";
            } else {
              label.classList.add("wrong-choice");
              tag.style.display = "inline-block";
              tag.className = "choice-feedback-tag tag-wrong";
              tag.textContent = "คำตอบที่คุณเลือก (ผิด)";
            }
          }
        });

        // Hiding explanation boxes to keep correct choices concealed
        const expBox = document.getElementById(`explanation-${q.id}`);
        if (expBox) expBox.classList.remove("visible");
      });

      const wrongCount = QUESTIONS.length - score;
      const percent = Math.round((score / QUESTIONS.length) * 100);

      // Human-readable time spent
      const spentMin = Math.floor(timeSpentSeconds / 60);
      const spentSec = timeSpentSeconds % 60;
      const spentStr = spentMin > 0 ? spentMin + " นาที " + spentSec + " วินาที" : spentSec + " วินาที";

      document.getElementById("resultStudentName").textContent =
        "ผู้เข้าสอบ: " + studentNameInput.value.trim();
      document.getElementById("resultStudentMeta").textContent =
        "ระดับชั้น/ห้อง: " + studentClassInput.value.trim() + "  |  รหัสนักศึกษา: " + studentNoInput.value.trim();
      document.getElementById("resultTimeMeta").textContent = 
        "⏱️ เวลาที่ใช้ทำข้อสอบ: " + spentStr + " (จากเวลาทั้งหมด 30 นาที)";
      
      document.getElementById("resultScoreText").textContent = score + " / " + QUESTIONS.length;
      document.getElementById("statCorrect").textContent = "ตอบถูก " + score + " ข้อ";
      document.getElementById("statIncorrect").textContent = "ตอบผิด " + wrongCount + " ข้อ";
      document.getElementById("statPercent").textContent = "คิดเป็นร้อยละ " + percent + "%";

      filterAllBtn.textContent = "ดูทุกข้อ (" + QUESTIONS.length + ")";
      filterWrongBtn.textContent = "ดูเฉพาะข้อที่ตอบผิด (" + wrongCount + ")";

      scoreBanner.classList.add("visible");
      
      // Keep submit button visible but state changed to indicate sync status
      submitBtn.disabled = true;
      submitBtn.style.display = "inline-flex";
      submitBtn.style.opacity = "0.7";
      submitBtn.textContent = "บันทึกผลคะแนนเสร็จสิ้น";

      const payload = {
        name: studentNameInput.value.trim(),
        className: studentClassInput.value.trim(),
        studentId: studentNoInput.value.trim(),
        score: score,
        timeSpent: spentStr,
        timestamp: new Date().toISOString()
      };

      const webAppUrl = "https://script.google.com/macros/s/AKfycbxkcdpQK3DVMNwXLpAVmsG-f05U-fTOTFyuLVgLRVPAiTQ1IV2FXdnNHfSyRH6qfFWh/exec";

      // Send to Google Sheets if deployed on Apps Script as a Web App
      if (typeof google !== 'undefined' && google.script && google.script.run) {
        submitBtn.textContent = "กำลังบันทึกคะแนนลง Google Sheet...";
        google.script.run
          .withSuccessHandler(() => {
            submitBtn.textContent = "✅ บันทึกคะแนนลง Sheet สำเร็จ!";
            setTimeout(() => {
              submitBtn.style.display = "none";
            }, 3000);
          })
          .withFailureHandler(() => {
            submitToSheetViaFetch(webAppUrl, payload);
          })
          .recordScore(payload);
      } else {
        // If running standalone (Offline/Directly) - Bypasses Google multi-account bug!
        submitToSheetViaFetch(webAppUrl, payload);
      }

      clearBtn.style.display = "none";
      retakeBtn.style.display = "inline-flex";

      scoreBanner.scrollIntoView({ behavior: "smooth", block: "start" });
    }

    function applyFilter(mode) {
      currentFilter = mode;
      filterAllBtn.classList.toggle("active", mode === "all");
      filterWrongBtn.classList.toggle("active", mode === "wrong");

      QUESTIONS.forEach(q => {
        const card = document.getElementById(`question-card-${q.id}`);
        if (mode === "all") {
          card.style.display = "block";
        } else {
          const isCorrect = card.dataset.correct === "true";
          card.style.display = isCorrect ? "none" : "block";
        }
      });
    }

    filterAllBtn.addEventListener("click", () => applyFilter("all"));
    filterWrongBtn.addEventListener("click", () => applyFilter("wrong"));

    function resetQuiz(keepStudentInfo = false) {
      isSubmitted = false;
      userAnswers = {};
      currentFilter = "all";

      if (!keepStudentInfo) {
        studentNameInput.value = "";
        studentClassInput.value = "";
        studentNoInput.value = "";
      }

      studentNameInput.disabled = false;
      studentClassInput.disabled = false;
      studentNoInput.disabled = false;
      studentNameInput.setAttribute("aria-invalid", "false");
      studentClassInput.setAttribute("aria-invalid", "false");
      studentNoInput.setAttribute("aria-invalid", "false");
      document.getElementById("nameError").classList.remove("visible");
      document.getElementById("classError").classList.remove("visible");
      document.getElementById("noError").classList.remove("visible");

      validationBanner.classList.remove("visible");
      scoreBanner.classList.remove("visible");

      submitBtn.style.display = "inline-flex";
      clearBtn.style.display = "inline-flex";
      retakeBtn.style.display = "none";

      filterAllBtn.classList.add("active");
      filterWrongBtn.classList.remove("active");

      renderQuestions();
      updateProgress();
      startTimer();
      window.scrollTo({ top: 0, behavior: "smooth" });
    }

    clearBtn.addEventListener("click", () => resetQuiz(true));
    retakeBtn.addEventListener("click", () => resetQuiz(true));

    function submitToSheetViaFetch(url, payload) {
      submitBtn.textContent = "กำลังส่งคะแนนไปยัง Google Sheet...";
      
      fetch(url, {
        method: "POST",
        mode: "no-cors", // Bypasses CORS and multi-account login check!
        cache: "no-cache",
        headers: {
          "Content-Type": "text/plain"
        },
        body: JSON.stringify(payload)
      })
      .then(() => {
        submitBtn.textContent = "✅ ส่งคะแนนเสร็จสิ้น!";
        setTimeout(() => {
          submitBtn.style.display = "none";
        }, 3000);
      })
      .catch((err) => {
        console.error("Fetch error:", err);
        submitBtn.textContent = "❌ บันทึกคะแนนล้มเหลว";
        submitBtn.disabled = false;
        submitBtn.style.opacity = "1";
      });
    }

    function setupShareFeature() {
      let currentUrl = window.location.href;
      
      // Fallback if sandboxed or empty
      if (currentUrl.includes("srcdoc") || currentUrl === "about:blank" || !currentUrl.startsWith("http")) {
        currentUrl = "about:invalid#zPythonSafez";
      }

      const shareUrlInput = document.getElementById("shareUrlInput");
      const copyUrlBtn = document.getElementById("copyUrlBtn");

      shareUrlInput.value = currentUrl;

      copyUrlBtn.addEventListener("click", () => {
        try {
          navigator.clipboard.writeText(currentUrl).then(() => {
            copyUrlBtn.innerHTML = "✅ คัดลอกลิงก์สำเร็จแล้ว!";
            setTimeout(() => {
              copyUrlBtn.innerHTML = "📋 คัดลอกลิงก์แบบทดสอบ";
            }, 2000);
          }).catch(() => {
            shareUrlInput.select();
            document.execCommand("copy");
            copyUrlBtn.innerHTML = "✅ คัดลอกลิงก์สำเร็จแล้ว!";
            setTimeout(() => {
              copyUrlBtn.innerHTML = "📋 คัดลอกลิงก์แบบทดสอบ";
            }, 2000);
          });
        } catch (e) {
          shareUrlInput.select();
          document.execCommand("copy");
          copyUrlBtn.innerHTML = "✅ คัดลอกลิงก์สำเร็จแล้ว!";
          setTimeout(() => {
            copyUrlBtn.innerHTML = "📋 คัดลอกลิงก์แบบทดสอบ";
          }, 2000);
        }
      });
    }

    // Initialize
    renderQuestions();
    updateProgress();
    startTimer();
    setupShareFeature();
  </script>
